function secondHighest(arr) { // Handle empty or single-element arrays if (arr.length <= 1) { return -Infinity; } // Handle arrays where all elements are the same let allSame = true; for (let i = 1; i < arr.length; i++) { if (Number(arr[i]) !== Number(arr[0])) { allSame = false; break; } } if (allSame) { return -Infinity; } // Find the second-highest distinct element let highest = -Infinity; let second = -Infinity; for (let i = 0; i < arr.length; i++) { let num = Number(arr[i]); if (num > highest) { second = highest; highest = num; } else if (num < highest && num > second) { second = num; } } return second; }
